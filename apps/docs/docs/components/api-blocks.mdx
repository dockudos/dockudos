<Endpoint method="POST" path="/v1/users" baseUrl="https://api.acme.dev" />

<ParamField name="email" type="string" location="body" required>
  The user's email address.
</ParamField>
<ParamField
  name="role"
  type="'admin' | 'member'"
  location="body"
  default="'member'"
>
  Access level of the new user.
</ParamField>
<ParamField name="X-Request-Id" type="string" location="header">
  Optional. Echoed back in the response so you can match logs.
</ParamField>

<ResponseField name="id" type="string">
  The id of the new user.
</ResponseField>
<ResponseField name="created_at" type="string (ISO 8601)">
  When the user was created.
</ResponseField>

```mdx
<Endpoint method="POST" path="/v1/users" baseUrl="https://api.acme.dev" />

  The user's email address.
</ParamField>

<ResponseField name="id" type="string">
  The id of the new user.
</ResponseField>
```
